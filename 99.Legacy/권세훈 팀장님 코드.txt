                                and not ip.endswith(".0")
                                and not ip.endswith(".255")]))

        arp_resp = self.execute_command("cat /proc/net/arp")
        arp_ips = re.findall(r'(\d+\.\d+\.\d+\.\d+)', arp_resp)
        for ip in arp_ips:
            if ip.startswith("172.18.0.") and ip not in candidates:
                candidates.append(ip)

        my_ip_resp = self.execute_command("cat /etc/hosts")
        my_ips = re.findall(r'(172\.18\.0\.\d+)', my_ip_resp)
        candidates = [ip for ip in candidates if ip not in my_ips]

        print(f"[*] All candidates: {candidates}")

        for ip in candidates:
            print(f"[*] Checking {ip}:8111 ...", end=" ", flush=True)
            result = self.execute_command(
